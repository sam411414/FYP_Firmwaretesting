#include <Arduino.h>
#include <Adafruit_NeoPixel.h>
#include <WiFi.h>
#include <esp_now.h>
#include "motorControl.h"

// Hardcoded MAC addresses for reliable connection
constexpr uint8_t kS3MacAddr[6] = {0x30, 0xED, 0xA0, 0x27, 0x8F, 0xA4}; // S3's MAC

// NeoPixel setup
constexpr int kLedPin = 8;  // GPIO8 for C3
constexpr int kLedCount = 1;
constexpr uint8_t kLedBrightness = 40;
Adafruit_NeoPixel ledStrip(kLedCount, kLedPin, NEO_GRB + NEO_KHZ800);

// ESP-NOW communication structures  
struct ControlPacket {
  uint8_t duty_cycle; // 40-100%
  uint8_t direction;  // 0=reverse, 1=forward
  uint8_t enable;     // 0=stop, 1=run
  uint8_t sync_blink; // 0=no blink, 1=sync blink
};

struct StatusPacket {
  uint8_t current_duty;
  uint8_t current_direction; 
  uint8_t motor_running;
};

// Global instances
MotorController motor;
bool g_connection_established = false;

// Synchronized blinking function
void performSyncBlink() {
  Serial.println("Synchronized blink starting...");
  for (int i = 0; i < 5; i++) {
    ledStrip.setPixelColor(0, ledStrip.Color(255, 0, 255)); // Magenta for sync
    ledStrip.show();
    delay(200);
    ledStrip.clear();
    ledStrip.show();
    delay(200);
  }
  Serial.println("Synchronized blink complete.");
}

// Send status back to S3
void sendStatusToS3() {
  StatusPacket status = {
    .current_duty = motor.getDutyCycle(),
    .current_direction = motor.getDirection() ? 1 : 0,
    .motor_running = 1  // Will be more sophisticated later
  };
  
  esp_now_send(kS3MacAddr, reinterpret_cast<uint8_t*>(&status), sizeof(status));
}

// ESP-NOW receive callback - handles commands from S3
void onDataRecv(const uint8_t *mac, const uint8_t *data, int len) {
  if (len != sizeof(ControlPacket)) {
    return;
  }

  // Connection established on first packet
  if (!g_connection_established) {
    g_connection_established = true;
    Serial.println("ESP-NOW connection established with S3");
  }

  // Parse incoming command
  ControlPacket cmd = {};
  memcpy(&cmd, data, sizeof(cmd));
  
  // Handle synchronized blink command
  if (cmd.sync_blink == 1) {
    performSyncBlink();
  }
  
  // Apply motor commands
  motor.setDutyCycle(cmd.duty_cycle);
  motor.setDirection(cmd.direction == 1);
  
  Serial.print("Received - Duty: ");
  Serial.print(cmd.duty_cycle);
  Serial.print("%, Dir: ");
  Serial.print(cmd.direction ? "Forward" : "Reverse");
  Serial.print(", Enable: ");
  Serial.println(cmd.enable ? "On" : "Off");
  
  // Send status back to S3
  sendStatusToS3();
}

void setup() {
  Serial.begin(115200);
  delay(200);

  // Initialize NeoPixel
  ledStrip.begin();
  ledStrip.setBrightness(kLedBrightness);
  ledStrip.clear();
  ledStrip.show();

  // Initialize motor controller hardware
  motor.initialize();

  // Initialize WiFi for ESP-NOW
  WiFi.mode(WIFI_STA);
  Serial.print("C3 MAC Address: ");
  Serial.println(WiFi.macAddress());

  // Initialize ESP-NOW
  if (esp_now_init() != ESP_OK) {
    Serial.println("ESP-NOW init failed");
    return;
  }

  // Register receive callback
  esp_now_register_recv_cb(onDataRecv);

  // Add S3 as peer for bidirectional communication
  esp_now_peer_info_t peer_info = {};
  memcpy(peer_info.peer_addr, kS3MacAddr, 6);
  peer_info.channel = 6;  // Match S3's channel
  peer_info.encrypt = false;

  if (esp_now_add_peer(&peer_info) != ESP_OK) {
    Serial.println("Failed to add S3 as peer");
    return;
  }

  Serial.println("C3 Motor Control Ready");
  Serial.print("Freq: ");
  Serial.print(motor.getFrequency());
  Serial.println(" Hz (fixed)");
  Serial.print("Initial duty: ");
  Serial.print(motor.getDutyCycle());
  Serial.println("%");
  Serial.println("Waiting for S3 connection...");
}

void loop() {
  // C3 board only responds to ESP-NOW commands from S3
  // No terminal input processing on C3
  delay(10); // Small delay for stability
}
