#include <Arduino.h>
#include <WiFi.h>
#include "motorControl.h"
#include "Encoder.h"
#include "ESPNOW_C3.h"

MotorController motor;
Encoder encoder(0, 1); // Use GPIO0 and GPIO1 for encoder A & B inputs

void setup() {
  Serial.begin(115200);
  delay(500);

  WiFi.mode(WIFI_STA);
  WiFi.disconnect();

  Serial.println("[C3] Motor test harness");
  Serial.print("MAC: ");
  Serial.println(WiFi.macAddress());

  // Init transport
  espnow_init();

  // Motor + encoder
  motor.initialize();
  encoder.init();
  motor.attachEncoder(&encoder);
  motor.setEnabled(false);
  espnow_register_motor(&motor);

  Serial.println("[C3] Motor + encoder registered, waiting for S3 commands...");
}

void loop() {
  motor.update();
  espnow_update();
  delay(10);
}
