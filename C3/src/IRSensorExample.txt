#include <Arduino.h>
#include <WiFi.h>
#include "IRSensor.h"
#include "ESPNOW_C3.h"

IRSensor ir_sensor(4, 3);  // DO: GPIO4, AO: GPIO3

void setup() {
  Serial.begin(115200);
  delay(500);

  WiFi.mode(WIFI_STA);
  WiFi.disconnect();

  Serial.println("[C3] IR sensor test harness");
  Serial.print("MAC: ");
  Serial.println(WiFi.macAddress());

  // Init transport
  espnow_init();

  // IR sensor
  ir_sensor.init();
  espnow_register_ir(&ir_sensor);

  Serial.println("[C3] IR sensor registered, waiting for S3 commands...");
}

void loop() {
  ir_sensor.update();
  espnow_update();
  delay(10);
}
