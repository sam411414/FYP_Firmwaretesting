// ═══════════════════════════════════════════════════════════════════
// Motor Test Harness (S3 side) — rename to main.cpp to use
// Commands: 0-100 (duty), F (forward), R (reverse), S (stop)
// Displays: [Motor] duty% DIR ON/OFF RPM
// ═══════════════════════════════════════════════════════════════════
#include <Arduino.h>
#include <WiFi.h>
#include "ESPNOW_S3.h"

namespace {
constexpr uint8_t kC3MacAddr[6] = {0xA0, 0x76, 0x4E, 0x7B, 0x9A, 0xB4};

uint8_t g_motor_duty   = 0;
uint8_t g_motor_dir    = 1;  // 1=forward
uint8_t g_motor_enable = 0;  // 0=stopped

void sendMotorCommand() {
  espnow_send_motor_control(g_motor_duty, g_motor_dir, g_motor_enable);
  espnow_set_motor_expected(g_motor_duty, g_motor_dir, g_motor_enable);
}

bool isNumeric(const String &s) {
  if (s.isEmpty()) return false;
  for (unsigned int i = 0; i < s.length(); i++) {
    if (!isDigit(s.charAt(i))) return false;
  }
  return true;
}
} // namespace

void setup() {
  Serial.begin(115200);
  delay(500);

  WiFi.mode(WIFI_STA);
  WiFi.disconnect();

  Serial.print("S3 MAC Address: ");
  Serial.println(WiFi.macAddress());

  espnow_init_sender(kC3MacAddr);

  Serial.println("=== Motor Test ===");
  Serial.println("0-100 : Set duty cycle (< 10 = stop)");
  Serial.println("F     : Forward");
  Serial.println("R     : Reverse");
  Serial.println("S     : Stop");
}

void loop() {
  espnow_process();

  if (!Serial.available()) {
    delay(10);
    return;
  }

  String line = Serial.readStringUntil('\n');
  line.trim();
  if (line.isEmpty()) return;

  // Numeric input → duty cycle
  if (isNumeric(line)) {
    int val = line.toInt();
    if (val > 100) val = 100;
    g_motor_duty = static_cast<uint8_t>(val);
    if (val < 10) {
      g_motor_enable = 0;
      g_motor_duty = 0;
    } else {
      g_motor_enable = 1;
    }
    sendMotorCommand();
    Serial.print("Motor duty -> ");
    Serial.println(g_motor_duty);
    return;
  }

  String cmd = line;
  cmd.toLowerCase();

  if (cmd == "f") {
    g_motor_dir = 1;
    sendMotorCommand();
    Serial.println("Motor -> FORWARD");
    return;
  }
  if (cmd == "r") {
    g_motor_dir = 0;
    sendMotorCommand();
    Serial.println("Motor -> REVERSE");
    return;
  }
  if (cmd == "s") {
    g_motor_enable = 0;
    g_motor_duty = 0;
    sendMotorCommand();
    Serial.println("Motor -> STOP");
    return;
  }

  Serial.println("Unknown. Use: 0-100 / F / R / S");
}
