#include <Arduino.h>
#include <Adafruit_NeoPixel.h>
#include <WiFi.h>
#include <esp_now.h>

namespace {
// Hardcoded MAC address for reliable C3 connection
constexpr uint8_t kC3MacAddr[6] = {0xA0, 0x76, 0x4E, 0x7B, 0x9A, 0xB4}; // C3's MAC

constexpr int kLedPin = 48;
constexpr int kLedCount = 1;
constexpr uint8_t kLedBrightness = 40;

Adafruit_NeoPixel ledStrip(kLedCount, kLedPin, NEO_GRB + NEO_KHZ800);

// Updated communication structures to match C3
struct ControlPacket {
  uint8_t duty_cycle; // 40-100%
  uint8_t direction;  // 0=reverse, 1=forward  
  uint8_t enable;     // 0=stop, 1=run
  uint8_t sync_blink; // 0=no blink, 1=sync blink
};

struct StatusPacket {
  uint8_t current_duty;
  uint8_t current_direction;
  uint8_t motor_running;
};

ControlPacket g_packet{ 50, 1, 1, 0 }; // Start with 50% duty, forward, enabled, no blink
StatusPacket g_c3_status{ 50, 1, 0 }; // Default C3 status

void onDataSent(const uint8_t *, esp_now_send_status_t status) {
  Serial.print("ESP-NOW send: ");
  Serial.println(status == ESP_NOW_SEND_SUCCESS ? "OK" : "FAIL");
}

void onDataRecv(const uint8_t *, const uint8_t *data, int len) {
  if (len == sizeof(StatusPacket)) {
    StatusPacket incoming = {};
    memcpy(&incoming, data, sizeof(incoming));
    g_c3_status = incoming;
    
    Serial.println("=== C3 Status ===");
    Serial.print("Duty: ");
    Serial.print(g_c3_status.current_duty);
    Serial.println("%");
    Serial.print("Direction: ");
    Serial.println(g_c3_status.current_direction ? "Forward" : "Reverse");
    Serial.print("Motor: ");
    Serial.println(g_c3_status.motor_running ? "Running" : "Stopped");
    Serial.println("=================");
  }
}

void blinkLed() {
  Serial.println("S3 solo blink...");
  for (int i = 0; i < 5; i++) {
    ledStrip.setPixelColor(0, ledStrip.Color(0, 255, 0)); // Green for solo
    ledStrip.show();
    delay(200);
    ledStrip.clear();
    ledStrip.show();
    delay(200);
  }
  Serial.println("S3 blink complete.");
}

void performSyncBlink() {
  Serial.println("Synchronized blink starting (S3+C3)...");
  for (int i = 0; i < 5; i++) {
    ledStrip.setPixelColor(0, ledStrip.Color(255, 0, 255)); // Magenta for sync
    ledStrip.show();
    delay(200);
    ledStrip.clear();
    ledStrip.show();
    delay(200);
  }
  Serial.println("Synchronized blink complete.");
}

void blinkConnectionConfirmed() {
  Serial.println("ESP-NOW connection established!");
  for (int i = 0; i < 5; i++) {
    ledStrip.setPixelColor(0, ledStrip.Color(0, 0, 255)); // Blue for connection
    ledStrip.show();
    delay(200);
    ledStrip.clear();
    ledStrip.show();
    delay(200);
  }
}

bool processCommand(String line) {
  line.trim();
  line.toLowerCase();

  if (line == "blink") {
    blinkLed();
    return true;
  }

  if (line == "sync") {
    g_packet.sync_blink = 1;
    Serial.println("Synchronized blink command sent to C3");
    esp_now_send(kC3MacAddr, reinterpret_cast<uint8_t*>(&g_packet), sizeof(g_packet));
    performSyncBlink(); // S3 also blinks simultaneously
    g_packet.sync_blink = 0; // Reset blink flag
    return true;
  }

  if (line == "forward") {
    g_packet.direction = 1;
    g_packet.enable = 1;
    Serial.println("Direction set to forward");
    esp_now_send(kC3MacAddr, reinterpret_cast<uint8_t*>(&g_packet), sizeof(g_packet));
    return true;
  }
  
  if (line == "reverse") {
    g_packet.direction = 0;
    g_packet.enable = 1;
    Serial.println("Direction set to reverse");
    esp_now_send(kC3MacAddr, reinterpret_cast<uint8_t*>(&g_packet), sizeof(g_packet));
    return true;
  }

  if (line == "stop") {
    g_packet.enable = 0;
    Serial.println("Motor stop requested");
    esp_now_send(kC3MacAddr, reinterpret_cast<uint8_t*>(&g_packet), sizeof(g_packet));
    return true;
  }

  // Check if it's a number for duty cycle control (40-100%)
  bool is_number = true;
  for (size_t i = 0; i < line.length(); ++i) {
    if (!isDigit(line[i])) {
      is_number = false;
      break;
    }
  }
  
  if (is_number && !line.isEmpty()) {
    const int value = line.toInt();
    if (value >= 40 && value <= 100) {
      g_packet.duty_cycle = static_cast<uint8_t>(value);
      g_packet.enable = 1;
      Serial.print("Duty cycle set to: ");
      Serial.print(g_packet.duty_cycle);
      Serial.println("%");
      esp_now_send(kC3MacAddr, reinterpret_cast<uint8_t*>(&g_packet), sizeof(g_packet));
      return true;
    } else {
      Serial.println("Number out of range. Use 40-100 for duty cycle.");
      return false;
    }
  }

  Serial.println("Unknown command: " + line);
  Serial.println("Available commands: blink, sync, forward, reverse, stop, 40-100");
  Serial.println("- blink: S3 LED only");
  Serial.println("- sync: Both S3 and C3 LEDs synchronized");
  return false;
}
} // namespace

void setup() {
  Serial.begin(115200);
  delay(200);

  ledStrip.begin();
  ledStrip.setBrightness(kLedBrightness);
  ledStrip.clear();
  ledStrip.show();

  // Use STA mode for direct MAC-to-MAC communication
  WiFi.mode(WIFI_STA);
  Serial.print("S3 MAC Address: ");
  Serial.println(WiFi.macAddress());

  if (esp_now_init() != ESP_OK) {
    Serial.println("ESP-NOW init failed");
    return;
  }

  esp_now_register_send_cb(onDataSent);
  esp_now_register_recv_cb(onDataRecv);

  // Add C3 as peer using hardcoded MAC address
  esp_now_peer_info_t peer_info = {};
  memcpy(peer_info.peer_addr, kC3MacAddr, sizeof(kC3MacAddr));
  peer_info.channel = 6;
  peer_info.encrypt = false;

  if (esp_now_add_peer(&peer_info) != ESP_OK) {
    Serial.println("ESP-NOW add C3 peer failed");
    return;
  }

  blinkConnectionConfirmed();
  Serial.println("S3 Motor Command Interface Ready!");
  Serial.println("Commands: blink, sync, forward, reverse, stop, 40-100 (duty %)");
  Serial.println("- blink: S3 only");
  Serial.println("- sync: Both S3 and C3 synchronized");
  Serial.println("Waiting for C3 connection...");
}

void loop() {
  if (!Serial.available()) {
    return;
  }

  String line = Serial.readStringUntil('\n');
  processCommand(line);
}
